REPLICATION SETUP:
──────────────────────────────────────────────────────────────────────────────

Example ReplicatedMergeTree Table:

CREATE TABLE events ON CLUSTER production
(
    event_id UInt64,
    event_time DateTime,
    event_type String
)
ENGINE = ReplicatedMergeTree(
    '/clickhouse/tables/{shard}/default/events',
    '{replica}'
)
ORDER BY event_id
PARTITION BY toYYYYMM(event_time);

Example Distributed Table:

CREATE TABLE events_distributed ON CLUSTER production
(
    event_id UInt64,
    event_time DateTime,
    event_type String
)
ENGINE = Distributed(production, default, events);
